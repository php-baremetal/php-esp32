# spi -- Baremetal\Spi\Bus and \Device plus the display/panel framework (§ twin of php_ext_i2c).
#
# The component always registers (a stable REQUIRES list keeps ESP-IDF's early-expansion pass happy),
# but the sources compile to empty translation units unless PHP_SPI_BUILD is defined -- which happens
# only when the project enables it (-DPHP_EXT_SPI=ON, from [extensions.spi] enabled = true).
#
# The selected drivers come as a semicolon list in PHP_SPI_DRIVERS (flash-tool passes it from
# [extensions.spi] drivers). For each <name> the build compiles drivers/<name>/*.c and expects it to
# define `const spi_driver_desc_t <name>_driver_desc`; a table of pointers is generated below.

set(_driver_srcs "")
set(_driver_incs "")
set(_gen "${CMAKE_CURRENT_BINARY_DIR}/drivers.gen.c")
set(_body "/* Generated by components/php_ext_spi/CMakeLists.txt -- do not edit. */\n")
string(APPEND _body "#include \"spi_driver_desc.h\"\n\n")

set(_names "")
if(DEFINED PHP_SPI_DRIVERS AND NOT PHP_SPI_DRIVERS STREQUAL "")
    string(REPLACE "," ";" _names "${PHP_SPI_DRIVERS}")
endif()

foreach(_n ${_names})
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/drivers/${_n}/${_n}.c")
        message(FATAL_ERROR "PHP_SPI_DRIVERS lists '${_n}' but drivers/${_n}/${_n}.c does not exist.")
    endif()
    file(GLOB _dc "drivers/${_n}/*.c")
    list(APPEND _driver_srcs ${_dc})
    list(APPEND _driver_incs "drivers/${_n}")
    string(APPEND _body "extern const spi_driver_desc_t ${_n}_driver_desc;\n")
endforeach()

string(APPEND _body "\nconst spi_driver_desc_t *const spi_drivers[] = {\n")
foreach(_n ${_names})
    string(APPEND _body "    &${_n}_driver_desc,\n")
endforeach()
string(APPEND _body "    0,\n};\n")
list(LENGTH _names _n_drivers)
string(APPEND _body "const size_t spi_driver_count = ${_n_drivers};\n")
file(WRITE "${_gen}" "${_body}")

idf_component_register(
    SRCS "src/registry.c" "src/php_spi.c" "src/php_spi_bus.c" "src/php_spi_device.c"
         "${_gen}" ${_driver_srcs}
    INCLUDE_DIRS "include" ${_driver_incs}
    # php for the engine headers; esp_driver_spi for the master API; esp_driver_gpio for reset/backlight
    # lines; esp_lcd for the panel drivers. The module_entry symbol is pulled from a data table in the
    # php component, out of --gc-sections' reach -> WHOLE_ARCHIVE.
    REQUIRES php esp_driver_spi esp_driver_gpio esp_lcd
    WHOLE_ARCHIVE
)

if(PHP_EXT_SPI)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PHP_SPI_BUILD=1)
    # The sources lean on PHP's macro-heavy headers; same relaxed flags as the engine.
    target_compile_options(${COMPONENT_LIB} PRIVATE -w -fpermissive)
endif()
