# led -- Baremetal\Led\Device plus a per-chip driver framework (WS2812, SK6812, ...), backed by the
# ESP-IDF led_strip component (RMT). The pin and pixel count are passed at construction.
#
# The component always registers (a stable REQUIRES list keeps ESP-IDF's early-expansion pass happy),
# but the sources compile to empty translation units unless PHP_LED_BUILD is defined -- which happens
# only when the project enables it (-DPHP_EXT_LED=ON, from [extensions.led] enabled = true). The
# module registration in internal_functions.c is gated on the same flag, so code and registration stay
# in step.
#
# The selected drivers come as a semicolon list in PHP_LED_DRIVERS (flash-tool passes it from
# [extensions.led] leds). For each <name> the build compiles drivers/<name>/<name>.c and expects it
# to define `const led_driver_desc_t <name>_driver_desc`; a table of pointers is generated below.

set(_driver_srcs "")
set(_gen "${CMAKE_CURRENT_BINARY_DIR}/drivers.gen.c")
set(_body "/* Generated by components/php_ext_led/CMakeLists.txt -- do not edit. */\n")
string(APPEND _body "#include \"led_driver_desc.h\"\n\n")

set(_names "")
if(DEFINED PHP_LED_DRIVERS AND NOT PHP_LED_DRIVERS STREQUAL "")
    string(REPLACE "," ";" _names "${PHP_LED_DRIVERS}")
endif()

foreach(_n ${_names})
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/drivers/${_n}/${_n}.c")
        message(FATAL_ERROR "PHP_LED_DRIVERS lists '${_n}' but drivers/${_n}/${_n}.c does not exist.")
    endif()
    list(APPEND _driver_srcs "drivers/${_n}/${_n}.c")
    string(APPEND _body "extern const led_driver_desc_t ${_n}_driver_desc;\n")
endforeach()

string(APPEND _body "\nconst led_driver_desc_t *const led_drivers[] = {\n")
foreach(_n ${_names})
    string(APPEND _body "    &${_n}_driver_desc,\n")
endforeach()
string(APPEND _body "    0,\n};\n")
list(LENGTH _names _n_drivers)
string(APPEND _body "const size_t led_driver_count = ${_n_drivers};\n")
file(WRITE "${_gen}" "${_body}")

idf_component_register(
    SRCS "src/led_core.c" "src/php_led.c" "${_gen}" ${_driver_srcs}
    INCLUDE_DIRS "include"
    # php for the engine headers; led_strip (managed component) for the RMT-driven strip API. The
    # module_entry symbol is pulled from a data table in the php component, out of --gc-sections'
    # reach -> WHOLE_ARCHIVE.
    REQUIRES php led_strip
    WHOLE_ARCHIVE
)

if(PHP_EXT_LED)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PHP_LED_BUILD=1)
    # The sources lean on PHP's macro-heavy headers; same relaxed flags as the engine.
    target_compile_options(${COMPONENT_LIB} PRIVATE -w -fpermissive)
endif()
