# Per-project C extensions.
#
# When the build is given -DPHP_PROJECT_EXTS_DIR (phpflash sets it to a project's ./firmware/exts),
# every subdirectory there is treated as a PHP extension: its *.c sources are compiled in, and it
# must define `zend_module_entry <dirname>_module_entry` (the same shape as components/php_ext_gpio).
# A small registration table is generated from the list and linked with WHOLE_ARCHIVE; main.c
# registers each entry after php_embed_init() via zend_startup_module().
#
# An extension that needs ESP-IDF components beyond the common hardware set below can list them, one
# component per line, in <name>/idf_requires.txt.

set(_ext_srcs "")
set(_ext_incs "")
set(_ext_names "")
set(_ext_extra_reqs "")

if(DEFINED PHP_PROJECT_EXTS_DIR AND NOT PHP_PROJECT_EXTS_DIR STREQUAL "")
    if(NOT IS_ABSOLUTE "${PHP_PROJECT_EXTS_DIR}")
        set(PHP_PROJECT_EXTS_DIR "${CMAKE_SOURCE_DIR}/${PHP_PROJECT_EXTS_DIR}")
    endif()
    if(EXISTS "${PHP_PROJECT_EXTS_DIR}")
        file(GLOB _subdirs LIST_DIRECTORIES true "${PHP_PROJECT_EXTS_DIR}/*")
        foreach(_d ${_subdirs})
            if(IS_DIRECTORY "${_d}")
                file(GLOB _c "${_d}/*.c")
                if(_c)
                    get_filename_component(_name "${_d}" NAME)
                    list(APPEND _ext_srcs ${_c})
                    list(APPEND _ext_incs "${_d}")
                    list(APPEND _ext_names "${_name}")
                    if(EXISTS "${_d}/idf_requires.txt")
                        file(STRINGS "${_d}/idf_requires.txt" _req)
                        list(APPEND _ext_extra_reqs ${_req})
                    endif()
                endif()
            endif()
        endforeach()
    endif()
endif()

# Generate the registration table. Always define the count (0 when there are none) so main.c's weak
# reference resolves; only define the array when there is at least one entry.
list(LENGTH _ext_names _ext_count)
set(_gen "${CMAKE_CURRENT_BINARY_DIR}/project_exts_register.c")
set(_body "/* Generated by components/php_project_exts/CMakeLists.txt -- do not edit. */\n")
string(APPEND _body "#include \"zend_modules.h\"\n\n")
foreach(_n ${_ext_names})
    string(APPEND _body "extern zend_module_entry ${_n}_module_entry;\n")
endforeach()
if(_ext_count GREATER 0)
    string(APPEND _body "\nzend_module_entry * const php_esp32_project_extensions[] = {\n")
    foreach(_n ${_ext_names})
        string(APPEND _body "    &${_n}_module_entry,\n")
    endforeach()
    string(APPEND _body "};\n")
endif()
string(APPEND _body "const int php_esp32_project_extension_count = ${_ext_count};\n")
file(WRITE "${_gen}" "${_body}")

if(_ext_extra_reqs)
    list(REMOVE_DUPLICATES _ext_extra_reqs)
endif()

idf_component_register(
    SRCS ${_ext_srcs} "${_gen}"
    INCLUDE_DIRS ${_ext_incs}
    # php for the engine headers; the common hardware components so a driver extension has its
    # headers; plus anything an extension asked for in its idf_requires.txt.
    REQUIRES php esp_driver_gpio esp_driver_i2c driver esp_timer ${_ext_extra_reqs}
    # The module entries are reached only through the generated table, itself referenced weakly from
    # main.c; force the whole archive in so --gc-sections can't drop them (same as php_ext_gpio).
    WHOLE_ARCHIVE
)

if(_ext_names)
    message(STATUS "Project PHP extensions: ${_ext_names}")
endif()

# Extension sources lean on PHP's macro-heavy headers; keep them under the same relaxed flags as
# php_ext_gpio and the engine sources.
target_compile_options(${COMPONENT_LIB} PRIVATE -w -fpermissive)
