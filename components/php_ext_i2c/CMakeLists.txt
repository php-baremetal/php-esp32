# i2c -- Baremetal\I2c\Bus and \Device plus the driver framework.
#
# The component always registers (a stable REQUIRES list keeps ESP-IDF's early-expansion pass happy),
# but the sources compile to empty translation units unless PHP_I2C_BUILD is defined -- which happens
# only when the project enables it (-DPHP_EXT_I2C=ON, from [extensions.i2c] enabled = true). The
# module registration in internal_functions.c is gated on the same flag, so code and registration stay
# in step.
#
# The selected drivers come as a semicolon list in PHP_I2C_DRIVERS (flash-tool passes it from
# [extensions.i2c] drivers). For each <name> the build compiles drivers/<name>/<name>.c and expects
# it to define `const i2c_driver_desc_t <name>_driver_desc`; a table of pointers is generated below.

set(_driver_srcs "")
set(_gen "${CMAKE_CURRENT_BINARY_DIR}/drivers.gen.c")
set(_body "/* Generated by components/php_ext_i2c/CMakeLists.txt -- do not edit. */\n")
string(APPEND _body "#include \"i2c_driver_desc.h\"\n\n")

set(_names "")
if(DEFINED PHP_I2C_DRIVERS AND NOT PHP_I2C_DRIVERS STREQUAL "")
    string(REPLACE "," ";" _names "${PHP_I2C_DRIVERS}")
endif()

foreach(_n ${_names})
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/drivers/${_n}/${_n}.c")
        message(FATAL_ERROR "PHP_I2C_DRIVERS lists '${_n}' but drivers/${_n}/${_n}.c does not exist.")
    endif()
    list(APPEND _driver_srcs "drivers/${_n}/${_n}.c")
    string(APPEND _body "extern const i2c_driver_desc_t ${_n}_driver_desc;\n")
endforeach()

string(APPEND _body "\nconst i2c_driver_desc_t *const i2c_drivers[] = {\n")
foreach(_n ${_names})
    string(APPEND _body "    &${_n}_driver_desc,\n")
endforeach()
string(APPEND _body "    0,\n};\n")
list(LENGTH _names _n_drivers)
string(APPEND _body "const size_t i2c_driver_count = ${_n_drivers};\n")
file(WRITE "${_gen}" "${_body}")

idf_component_register(
    SRCS "src/registry.c" "src/php_i2c.c" "src/php_i2c_bus.c" "src/php_i2c_device.c"
         "${_gen}" ${_driver_srcs}
    INCLUDE_DIRS "include"
    # php for the engine headers; esp_driver_i2c for the master-bus API; esp_driver_gpio for the
    # drivers that own a reset/interrupt line (e.g. the touch controller). The module_entry symbol is
    # pulled from a data table in the php component, out of --gc-sections' reach -> WHOLE_ARCHIVE.
    REQUIRES php esp_driver_i2c esp_driver_gpio
    WHOLE_ARCHIVE
)

if(PHP_EXT_I2C)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PHP_I2C_BUILD=1)
    # The sources lean on PHP's macro-heavy headers; same relaxed flags as the engine.
    target_compile_options(${COMPONENT_LIB} PRIVATE -w -fpermissive)
endif()
