# WebSocket at /ws (echoed back; a minimal client does the handshake + masked frames):
handshake: HTTP/1.1 101 Switching Protocols

# HTTP and WebSocket share one server + one reactor queue; /ws wins over the catch-all.

# ws_broadcast(): a 1 Hz timer pushes a heartbeat to every connected client (unsolicited). Two clients
# connected at once both receive the same frames:
#   client A (listen only): {"beat":36} {"beat":37} {"beat":38}
#   client B (listen only): {"beat":36} {"beat":37} {"beat":38}
# ...while another client's echo (reply) keeps working on the same server.
